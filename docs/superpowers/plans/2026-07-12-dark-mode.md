# Dark Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a three-state (System/Light/Dark) dark mode to Mosaic Builder that re-themes all editor chrome while keeping figure content — and therefore exports — byte-identical across themes.

**Architecture:** Two layers. (A) A CSS-token override pair in `styles.css` driven by a `data-theme` attribute on `<html>` plus `prefers-color-scheme`; everything using `var(--c-*)` re-themes for free. (B) A chrome-only `theme` flag threaded through the existing `DrawFigureOpts` into the canvas draw path — only `drawPlaceholder` reads it; figure content never does. A `useTheme` hook + a synchronous boot stamp in `main.tsx` own the runtime and persist the choice to `localStorage`.

**Tech Stack:** React 18, TypeScript (strict, type-checked ESLint), Vitest (`node` env + recorder-based `DrawCtx` fakes), Electron renderer, plain CSS custom properties. No new dependencies.

## Global Constraints

- **Toolchain:** Node 22, pnpm 10.
- **Every step ends green on:** `pnpm typecheck && pnpm test && pnpm lint` (ESLint is `--max-warnings 0`; Prettier must pass).
- **No new dependencies.** No `zod`, no `electron-store`, no new IPC channel. Persistence is `localStorage` only.
- **No reskin:** radii (`--r-*`), type scale, spacing, and the accent hue's identity (`#2e5bff`) are unchanged. Dark mode is a second token set of the same system.
- **CSP is `script-src 'self'`** — no inline `<script>` in `index.html`. The boot theme stamp runs at the top of the renderer entry module (`main.tsx`).
- **Content vs chrome invariant:** figure content (canvas background, gutter color, letters, scale bars, annotations, and the fallback hexes that seed user-editable colors) is theme-independent. Only chrome re-themes. Exports must be byte-identical across themes.
- **`localStorage` key:** `"mosaic-theme"`. Values: `"system" | "light" | "dark"`.
- **Unused vars must be prefixed `_`.** Reducer-style exhaustiveness guards use `const _x: never = …`.
- **Test location:** `test/unit/*.test.ts(x)`, one file per module.

```ts
export type ThemeChoice = "system" | "light" | "dark";
export type EffectiveTheme = "light" | "dark";

export const THEME_STORAGE_KEY = "mosaic-theme";

export function resolveEffectiveTheme(
  choice: ThemeChoice,
  prefersDark: boolean,
): EffectiveTheme {
  if (choice === "system") return prefersDark ? "dark" : "light";
  return choice;
}

function isThemeChoice(v: unknown): v is ThemeChoice {
  return v === "system" || v === "light" || v === "dark";
}

export function readStoredChoice(): ThemeChoice {
  try {
    const raw = localStorage.getItem(THEME_STORAGE_KEY);
    return isThemeChoice(raw) ? raw : "system";
  } catch {
    // localStorage can throw in locked-down contexts; fall back to system.
    return "system";
  }
}

export function writeStoredChoice(choice: ThemeChoice): void {
  try {
    localStorage.setItem(THEME_STORAGE_KEY, choice);
  } catch {
    // Persistence is best-effort; a failed write must not break theming.
  }
}

export function applyThemeAttr(choice: ThemeChoice): void {
  const root = document.documentElement;
  // "system" carries NO attribute so the prefers-color-scheme media query
  // is what drives the tokens; an explicit choice stamps the override.
  if (choice === "system") {
    root.removeAttribute("data-theme");
  } else {
    root.setAttribute("data-theme", choice);
  }
}
```

```ts
function drawPlaceholder(
  ctx: DrawCtx,
  rect: PanelRect,
  label: string,
  showCaption: boolean,
  theme: "light" | "dark" = "light",
): void {
  const dark = theme === "dark";
  ctx.fillStyle = dark ? PLACEHOLDER_FILL_DARK : PLACEHOLDER_FILL;
  ctx.fillRect(rect.x, rect.y, rect.w, rect.h);
  ctx.strokeStyle = dark ? PLACEHOLDER_STROKE_DARK : PLACEHOLDER_STROKE;
  ctx.strokeRect(rect.x + 0.5, rect.y + 0.5, rect.w - 1, rect.h - 1);

  const cx = rect.x + rect.w / 2;
  const cy = rect.y + rect.h / 2;
  ctx.fillStyle = dark ? PLACEHOLDER_LABEL_DARK : PLACEHOLDER_LABEL;
  ctx.font = "bold 24px system-ui, sans-serif";
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  ctx.fillText(label, cx, showCaption ? cy - 14 : cy);

  if (showCaption) {
    ctx.fillStyle = dark ? PLACEHOLDER_CAPTION_DARK : PLACEHOLDER_CAPTION;
    ctx.font = "13px system-ui, sans-serif";
    ctx.fillText(DRAG_HINT_TEXT, cx, cy + 14);
  }
}
```

```ts
import { useCallback, useEffect, useState } from "react";
import {
  applyThemeAttr,
  readStoredChoice,
  resolveEffectiveTheme,
  writeStoredChoice,
  type EffectiveTheme,
  type ThemeChoice,
} from "./theme";

const DARK_QUERY = "(prefers-color-scheme: dark)";

function prefersDark(): boolean {
  return (
    typeof matchMedia === "function" && matchMedia(DARK_QUERY).matches === true
  );
}

export interface UseTheme {
  choice: ThemeChoice;
  effectiveTheme: EffectiveTheme;
  setChoice(next: ThemeChoice): void;
}

export function useTheme(): UseTheme {
  const [choice, setChoiceState] = useState<ThemeChoice>(() =>
    readStoredChoice(),
  );
  const [osDark, setOsDark] = useState<boolean>(() => prefersDark());

  // When following the OS, re-render if the OS theme flips while we're open.
  // When pinned, the listener is attached but osDark simply doesn't feed the
  // resolved theme, so it's inert.
  useEffect(() => {
    if (typeof matchMedia !== "function") return;
    const mq = matchMedia(DARK_QUERY);
    const onChange = (e: MediaQueryListEvent): void => setOsDark(e.matches);
    mq.addEventListener("change", onChange);
    return () => mq.removeEventListener("change", onChange);
  }, []);

  // Keep the <html> attribute in sync with the current choice (covers the
  // mount case too, reinforcing the boot stamp in main.tsx).
  useEffect(() => {
    applyThemeAttr(choice);
  }, [choice]);

  const setChoice = useCallback((next: ThemeChoice): void => {
    setChoiceState(next);
    writeStoredChoice(next);
    applyThemeAttr(next);
  }, []);

  const effectiveTheme = resolveEffectiveTheme(choice, osDark);
  return { choice, effectiveTheme, setChoice };
}
```
